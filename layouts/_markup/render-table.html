{{- /* Tables scroll inside their own frame on narrow screens. {cols="1,3"} on the line after a table sets the relative width of each column, as AsciiDoc's cols attribute did. The percentages are computed like Asciidoctor: rounded to 4 decimals, the last column takes the rest. */ -}} {{- $widths := slice -}} {{- with .Attributes.cols -}} {{- $weights := split . "," -}} {{- $columns := len (index (or $.THead $.TBody) 0) -}} {{- if ne (len $weights) $columns -}} {{- errorf "%s: a table has %d columns, but its cols attribute %q has %d values" $.Page.File.Path $columns . (len $weights) -}} {{- end -}} {{- $total := 0.0 -}} {{- range $weights }}{{ $total = add $total (float (strings.TrimSpace .)) }}{{ end -}} {{- $used := 0.0 -}} {{- range $i, $w := $weights -}} {{- $pct := div (math.Round (mul (div (float (strings.TrimSpace $w)) $total) 1000000)) 10000.0 -}} {{- if eq (add $i 1) (len $weights) }}{{ $pct = div (math.Round (mul (sub 100.0 $used) 10000)) 10000.0 }}{{ end -}} {{- $used = add $used $pct -}} {{- $widths = $widths | append $pct -}} {{- end -}} {{- end -}}
{{- with $widths }} {{ range . }}{{ end }} {{- end }} {{- range .THead }} {{- range . }} {{ .Text }} {{- end }} {{- end }} {{- range .TBody }} {{- range . }} {{ .Text }} {{- end }} {{- end }}
